Method Endpoint Deskripsi Autentikasi

package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"net/http"
)

func SendWhatsApp(nomor, pesan string) error {
	payload, _ := json.Marshal(map[string]string{
		"receiver": nomor,
		"message":  pesan,
	})

	req, err := http.NewRequest("POST", "http://localhost:8080/api/send/text", bytes.NewBuffer(payload))
	if err != nil {
		return err
	}
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("x-api-key", "mysecretkey123")

	client := &http.Client{}
	resp, err := client.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()

	fmt.Println("Response Status:", resp.Status)
	return nil
}

const axios = require('axios');

async function sendWhatsApp(receiver, message) {
  const res = await axios.post('http://localhost:8080/api/send/text', {
    receiver: receiver,
    message: message
  }, {
    headers: {
      'Content-Type': 'application/json',
      'x-api-key': 'mysecretkey123'
    }
  });

  return res.data;
}
import requests

def send_whatsapp(receiver, message):
    url = "http://localhost:8080/api/send/text"
    headers = {
        "Content-Type": "application/json",
        "x-api-key": "mysecretkey123"
    }
    payload = {
        "receiver": receiver,
        "message": message
    }
    response = requests.post(url, json=payload, headers=headers)
    return response.json()
